数据库创建：
create database qtsql;


创建表
use qtsql;

DROP TABLE IF EXISTS `qtdata`;
CREATE TABLE qtdata (
    id INT AUTO_INCREMENT PRIMARY KEY,
    curren_time VARCHAR(255),
    temp float(10,2) DEFAULT NULL,
    humi float(10,2) DEFAULT NULL,
    light float(10,2) DEFAULT NULL,
    soil float(10,2) DEFAULT NULL,
    mq2 float(10,2) DEFAULT NULL,
    rain float(10,2) DEFAULT NULL
)ENGINE=InnoDB  DEFAULT CHARSET=utf8;
